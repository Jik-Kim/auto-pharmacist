cmake_minimum_required(VERSION 3.16)
project(gmp_dsr_controller LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(dsr_controller2 REQUIRED)
find_package(gmp_interfaces REQUIRED)
find_package(pluginlib REQUIRED)
find_package(controller_interface REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)

add_library(${PROJECT_NAME} SHARED src/collision_controller.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PRIVATE include)
# 벤더와 동일한 DRCF_VERSION·헤더·라이브러리를 이어받는다. 별도 연결은 만들지 않는다.
target_link_libraries(${PROJECT_NAME} PUBLIC dsr_controller2::dsr_controller2)
ament_target_dependencies(${PROJECT_NAME} PUBLIC gmp_interfaces pluginlib controller_interface rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(controller_interface gmp_dsr_controller_plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_collision_read test/test_collision_read.cpp)
  target_include_directories(test_collision_read PRIVATE include)
  target_compile_features(test_collision_read PRIVATE cxx_std_17)
  ament_add_gtest(test_plugin_load test/test_plugin_load.cpp)
  ament_target_dependencies(test_plugin_load pluginlib controller_interface)
endif()
ament_package()
